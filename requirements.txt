# ThinkSpark-v2-350M — end-to-end (data-gen + Phase-0/1/2 training + inference).
# Tested target: Kaggle 2xT4 (CUDA), `conda activate llms`.
#
#   conda activate llms
#   pip install -r requirements.txt
#
# On Kaggle, torch/transformers are usually preinstalled; pip will no-op or upgrade.

# ---- core / training --------------------------------------------------------
torch>=2.2                 # bf16 + CUDA on 2xT4 (MPS/CPU also work, slower)
transformers>=4.44,<5      # Gemma-3-270M (google/gemma-3-270m) + MimiModel (kyutai/mimi)
# UPPER BOUND REQUIRED — real observed break: transformers 5.x raised its torch floor to
# 2.5, and just silently DISABLES its own PyTorch backend (not an install error — an
# ImportError only surfaced later, at model-load time: "PyTorch was not found") on any
# box with torch<2.5, e.g. a RunPod image pinned to torch 2.4.1+cu124 to match its baked
# CUDA build. Pinning transformers is far safer here than upgrading torch, which risks
# breaking that CUDA-matched pre-baked build instead.
accelerate>=0.33           # device placement / DDP helpers
peft>=0.11                 # LoRA for Phase-1 (optional; only if use_lora=true)
numpy>=1.26
pyyaml>=6.0
tqdm>=4.66
plotext>=5.2               # terminal loss curves (nice on Kaggle)

# ---- Phase-0 encode (Mimi) --------------------------------------------------
torchaudio>=2.2            # f0 extraction (falls back to numpy autocorrelation)
soundfile>=0.12            # robust wav io (optional; stdlib wave is the fallback)

# ---- Phase-1 corpus fetch (scripts/P1_01_fetch_corpus.py) -------------------
datasets>=2.19             # streams Common Voice / Kathbath / Shrutilipi / FLEURS / IndicTTS
# NO torchcodec dependency, deliberately — this project fetches Audio-cast columns with
# `Audio(decode=False)` and decodes the raw bytes itself via plain `soundfile` (already a
# dependency above), instead of using `datasets`' built-in decode path. Real, repeated
# breakage this avoids: `datasets` 4.x/5.x hard-requires torchcodec for its own decode
# path, but torchcodec categorically doesn't support torch<2.5 (no torchcodec version
# fixes that — confirmed against its own compatibility table, hit on a real box running
# torch 2.4.1), and separately hit CUDA-version-specific `libnvrtc`/`libavutil` shared-
# library failures even when a nominally-compatible version WAS installed (torchcodec
# 0.11+ hard-requires CUDA 13's libnvrtc regardless of GPU usage; older versions need
# system FFmpeg's exact matching .so version). None of that is needed for this project's
# actual use — decoding small audio clips, which `soundfile` already does reliably with
# zero torch/CUDA coupling.
